metricaNombre = 'SIG-SC-02 â€” Problemas reportados por el cliente';

  }

mostrarErrores = false;

constructor(
  private route: ActivatedRoute, 
  private router: Router, 
  private metricaService: MetricaAcademicaService, 
  private sprintService: SprintService
) {}

ngOnInit(): void {
  this.metricaId = this.route.snapshot.paramMap.get('id') || '';
  this.cargarContexto();
}

cargarContexto(): void {
  const proyectoStr = localStorage.getItem('mpdia_proyecto_activo');
  const sprintStr = localStorage.getItem('mpdia_sprint_activo');
  if (!proyectoStr) { 
    this.error = 'No hay un proyecto activo'; 
    this.loading = false; 
    return; 
  }
  if (!sprintStr) { 
    this.error = 'No hay un sprint activo'; 
    this.loading = false; 
    return; 
  }
  this.proyectoActivo = JSON.parse(proyectoStr);
  this.sprintActivo = JSON.parse(sprintStr);
  this.cargarParametrizacion();
}

cargarParametrizacion(): void {
  // TODO: Implementar cuando exista endpoint
  this.loading = false;
  this.sinParametrizacion = true;
}

esValido(nombre: string): boolean {
  const v = this.valores[nombre];
  return v !== null && v !== undefined && v !== '' && v >= 0 && Number.isInteger(Number(v));
}

ejecutar(): void {
  this.mostrarErrores = true;
  if (!this.esValido('problemas_reportados')) return;
  if (!this.proyectoActivo || !this.sprintActivo) return;
  
  this.ejecutando = true;
  this.errorEjecucion = '';
  
  this.metricaService.ejecutar(this.metricaId, {
    proyectoId: this.proyectoActivo.id, 
    sprintId: this.sprintActivo.id, 
    valores: this.valores
  }).subscribe({
    next: (r) => { 
      this.resultado = r; 
      this.ejecutando = false; 
      this.cargarHistorico(); 
    },
    error: (e) => { 
      this.ejecutando = false; 
      this.errorEjecucion = e.status === 403 ? 'Sin permiso' : 
                            e.status === 409 ? 'ParametrizaciÃ³n no aprobada' : 
                            'Error al ejecutar'; 
    }
  });
}

cargarHistorico(): void {
  if (!this.proyectoActivo) return;
  this.metricaService.obtenerHistorico(this.metricaId, this.proyectoActivo.id).subscribe({
    next: (h) => this.historico = h, 
    error: () => {}
  });
}

analizarConIA(): void {
  if (!this.resultado) return;
  
  this.interpretando = true;
  this.errorInterpretacion = '';
  
  this.metricaService.solicitarInterpretacion(this.resultado.resultadoId).subscribe({
    next: (i) => { 
      this.interpretacion = i; 
      this.interpretando = false; 
    },
    error: (e) => { 
      this.interpretando = false; 
      this.errorInterpretacion = e.status === 403 ? 'Sin permiso' : 'Error al interpretar'; 
    }
  });
}

irAParametrizacion(): void { 
  this.router.navigate(['/parametrizacion', this.metricaId]); 
}

volver(): void { 
  this.router.navigate(['/planeacion']); 
}
